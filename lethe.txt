local function find(...)

local function find_any(paths)
    for _, p in ipairs(paths) do
        local r = find(unpack(p))
        if r then return r end
    end
    return nil
end

-- override с пропуском несуществующих пунктов; без аргументов = сброс
local function ovr(ref, ...)
    if ref then pcall(ref.override, ref, ...) end
end

local function safe(fn, default)
    local ok, v = pcall(fn)
    if ok and v ~= nil then return v end
    return default
end

local function lerp(a, b, t) return a + (b - a) * t end
local function clamp(v, lo, hi) return math.max(lo, math.min(hi, v)) end
local function norm_yaw(y) return (y + 180) % 360 - 180 end

local function has(list, value)
    if type(list) ~= "table" then return false end
    for _, v in ipairs(list) do
        if v == value then return true end
    end
    return false
end

local function index_of(list, value)
    for i, v in ipairs(list) do
        if v == value then return i end
    end
    return 1
end

local function rint(lo, hi)
    lo, hi = math.floor(lo), math.floor(hi)
    if lo > hi then lo, hi = hi, lo end
    return math.random(lo, hi)
end

local function shuffle(list)
    for i = #list, 2, -1 do
        local j = math.random(i)
        list[i], list[j] = list[j], list[i]
    end
    return list
end

-- цвет -> escape-код (\aRRGGBBAA)
local function esc(c, a)
    return string.format("\a%02X%02X%02X%02X", math.floor(c.r), math.floor(c.g), math.floor(c.b), math.floor(a or c.a))
end

local function gradient_text(text, c1, c2, speed)
    local out, t = "", globals.curtime
    for i = 1, #text do
        local wave = clamp(math.cos(8 * speed * t + (i - 1) * 10 / 30), 0, 1)
        out = out .. string.format("\a%02X%02X%02X%02X%s",
            math.floor(lerp(c1.r, c2.r, wave)), math.floor(lerp(c1.g, c2.g, wave)),
            math.floor(lerp(c1.b, c2.b, wave)), math.floor(c1.a), text:sub(i, i))
    end
    return out
end

local function text_width(text, font)
    local plain = (tostring(text):gsub("\a%x%x%x%x%x%x%x%x", ""))
    return render.measure_text(font or 1, "", plain).x
end

local function draw_text(x, y, text, align, col, font)
    col = col or color(255, 255, 255, 255)
    if align == "c" then x = x - text_width(text, font) / 2
    elseif align == "r" then x = x - text_width(text, font) end
    render.text(font or 1, vector(x, y), col, "", text)
end

-- Паттерны значений для defensive
local function pattern(kind, v1, v2, speed, tick)
    speed = math.max(1, speed or 1)
    if kind == "None" then return 0
    elseif kind == "Up" then return -89
    elseif kind == "Down" then return 89
    elseif kind == "Zero" then return 0
    elseif kind == "Static" or kind == "Custom" then return v1
    elseif kind == "Swap" then return (tick % 2 == 0) and v1 or v2
    elseif kind == "Switch" then return (math.floor(tick / speed) % 2 == 0) and v1 or v2
    elseif kind == "Random" then return v1 + math.random() * (v2 - v1)
    elseif kind == "3-way" then
        local i = tick % 3
        return i == 0 and v1 or (i == 1 and (v1 + v2) / 2 or v2)
    elseif kind == "Sway" then return lerp(v1, v2, (math.sin(globals.curtime * speed) + 1) / 2)
    elseif kind == "Spinbot" then return norm_yaw(globals.curtime * speed * 90)
    end
    return v1
end

local uid = 0
local function uniq(name)
    uid = uid + 1
    local n, s = uid, ""
    while n > 0 do
        s = s .. ((n % 2 == 1) and "\r" or "\a00000000")
        n = math.floor(n / 2)
    end
    return name .. s
end

local function gear_of(el, fallback)
    local ok, g = pcall(function() return el:create() end)
    if ok and g then return g end
    return fallback
end

local function sl(g, name, lo, hi, def, suf) return g:slider(uniq(name), lo, hi, def, 1, suf or "") end
local function cb(g, name, list) return g:combo(uniq(name), list) end
local function tog(g, name, def) return g:switch(uniq(name), def or false) end
local function ms(g, name, list) return g:selectable(uniq(name), list) end

local function sel_set(el, list)
    pcall(el.set, el, list)
    local got = safe(function() return el:get() end, {})
    if type(got) ~= "table" or #got == 0 then pcall(el.set, el, unpack(list)) end
end

local function vis(el, state)
    if not el then return end
    if el.visibility then el:visibility(state and true or false)
    elseif el.set_visible then el:set_visible(state and true or false) end
end

local function is_el(v)
    local t = type(v)
    if t == "number" or t == "string" or t == "boolean" or t == "function" then return false end
    return safe(function() return v.get ~= nil and v.set ~= nil end, false)
end

    s.freeze_cd     = sl(gf, "Cooldown", 0, 40, 0, "t")
    s.freeze_side   = cb(gf, "Freeze on", {"Current", "Left", "Right", "Random"})

    function s.refresh()
        local en = s.shown and s.enable:get()
        vis(s.enable, s.shown)
        aa_vis(s, en)
        local f = en and s.freeze:get()
        vis(s.freeze, en)
        vis(s.freeze_chance, f); vis(s.freeze_min, f); vis(s.freeze_max, f); vis(s.freeze_cd, f); vis(s.freeze_side, f)

    s.enable:set_callback(s.refresh)
    s.freeze:set_callback(s.refresh)
    for _, k in ipairs(AA_KEYS) do s[k]:set_callback(s.refresh) end
    return s
end

    local s = {shown = false}
    s.enable = g_def:switch(uniq("Enable " .. title), false)

    s.pitch = g_def:combo(uniq("Pitch"), PITCH_KINDS)
    local gp = gear_of(s.pitch, g_def)
    s.p1   = sl(gp, "Value 1", -89, 89, 0, "°")
    s.p2   = sl(gp, "Value 2", -89, 89, 0, "°")
    s.pspd = sl(gp, "Speed", 1, 14, 1, "")

    s.yaw = g_def:combo(uniq("Yaw"), YAW_KINDS)
    local gy = gear_of(s.yaw, g_def)
    s.y1   = sl(gy, "Value 1", -180, 180, 0, "°")
    s.y2   = sl(gy, "Value 2", -180, 180, 0, "°")
    s.yspd = sl(gy, "Speed", 1, 14, 1, "")

    s.mod = g_def:combo(uniq("Modifier"), MOD_KINDS)
    local gm = gear_of(s.mod, g_def)
    s.m1   = sl(gm, "Value 1", -180, 180, 0, "°")
    s.m2   = sl(gm, "Value 2", -180, 180, 0, "°")
    s.mspd = sl(gm, "Speed", 1, 14, 1, "")

    -- Delay: как часто меняются значения Swap / Switch / 3-way внутри defensive-окна
    s.dl_mode = g_def:combo(uniq("Delay mode"), {"Static", "Random", "Shuffle"})
    s.dl_min  = sl(g_def, "Delay", 1, 14, 1, "t")
    s.dl_max  = sl(g_def, "Delay max", 1, 14, 1, "t")

    add_body(g_def, s)

    function s.refresh()
        local en = s.shown and s.enable:get()
        vis(s.enable, s.shown)
        vis(s.pitch, en); vis(s.yaw, en); vis(s.mod, en)
        vis(s.dl_mode, en); vis(s.dl_min, en); vis(s.dl_max, en and s.dl_mode:get() ~= "Static")
        body_vis(s, en)
    end
    for _, k in ipairs({"enable", "dl_mode", "body", "body_state", "body_limits"}) do
        s[k]:set_callback(s.refresh)
    end
    return s
end

local function snapshot(s)
    local vals = {}
    for k, v in pairs(s) do
        if is_el(v) then vals[k] = v:get() end
    end
    return vals
end

local function restore(s, vals)
    for k, v in pairs(vals) do
        if is_el(s[k]) then pcall(s[k].set, s[k], v) end
    end
    s.refresh()
end

local function get_state(cmd, lp)
    local speed = lp.m_vecVelocity:length2d()
    local on_ground = bit.band(lp.m_fFlags, 1) == 1
    local air_now = (not on_ground) or cmd.in_jump
    local fd = R.fakeduck and R.fakeduck:get()
    local duck = lp.m_flDuckAmount > 0.7 or fd
    local slow = R.slowwalk and R.slowwalk:get()

    local raw
    if air_now and duck then raw = 6
    elseif air_now then raw = 5
    elseif duck and speed > 10 then raw = 8
    elseif duck then raw = 7
    elseif slow and speed > 10 then raw = 3
    elseif speed > 5 then raw = 4
    else raw = 2 end
    return raw, on_ground
end

local function state_ok(sel, raw) return has(sel:get(), conds[raw]) end

local function resolve(sets, team_num, raw)
    local function try(t)
        if not t then return nil end
        local st = sets[t]
        if st[raw].enable:get() then return st[raw] end
        if st[1].enable:get() then return st[1] end
        return nil
    end
    local tn = (team_num == 2 and "T") or (team_num == 3 and "CT") or nil
    return try(tn) or try("Both")
end

-- ========== DEFENSIVE-ДВИЖОК ==========
local function track_defensive(lp)
    local tb = lp.m_nTickBase
    if math.abs(tb - def.max) > 64 then def.max = 0 end
    if tb > def.max then def.max = tb end
    def.ticks = clamp(def.max - tb - 1, 0, 14)

    def.exploit = ((R.dt and R.dt:get()) or (R.hs and R.hs:get())) and true or false
    local was = def.active
    def.active = def.exploit and def.ticks > 0 and def.ticks <= DEF_WINDOW

    if def.active and not was then
        def.age = 0; def.windows = def.windows + 1
    elseif def.active then
        def.age = def.age + 1
    else
        def.age = 0
    end
end

-- ========== ДВИЖОК СТОРОН (Side / Delay / Freeze / Anti-bruteforce) ==========

-- выбор значения из диапазона: Random (без повтора подряд) или Shuffle (мешок без повторов)
local function pick_range(lo, hi, key, mode, avoid)
    lo, hi = math.floor(lo), math.floor(hi)
    if lo > hi then lo, hi = hi, lo end
    if lo == hi then return lo end

    if mode == "Shuffle" then
        local bk = key .. ":" .. lo .. ":" .. hi
        local b = eng.bags[key]
        if not b or b.key ~= bk or #b.list == 0 then
            local list = {}
            for v = lo, hi do list[#list + 1] = v end
            b = {key = bk, list = shuffle(list)}
            eng.bags[key] = b
        end
        return table.remove(b.list)
    end

    local v = math.random(lo, hi)
    if avoid then
        for _ = 1, 3 do
            if v ~= eng.lastd[key] then break end
            v = math.random(lo, hi)
        end
    end
    eng.lastd[key] = v
    return v
end

-- сколько тиков держать сторону. left = сторона, на которую только что перешли
local function side_wait(bs, left)
    if not (bs and bs.delay:get()) then return 1 end
    local mode = bs.delay_mode:get()
    if mode == "Static" then
        return left and bs.delay_left:get() or bs.delay_right:get()
    end
    local lmin, lmax = bs.delay_lmin:get(), bs.delay_lmax:get()
    local rmin, rmax = bs.delay_rmin:get(), bs.delay_rmax:get()
    local mir = bs.delay_mirror:get()
    if mir == "Left = Right" then rmin, rmax = lmin, lmax
    elseif mir == "Right = Left" then lmin, lmax = rmin, rmax end
    local avoid = bs.delay_avoid:get()
    if left then return pick_range(lmin, lmax, "L", mode, avoid) end
    return pick_range(rmin, rmax, "R", mode, avoid)
end

-- выбор следующей стороны: Alternate / Random (с шансом и лимитом серии) / Bag (2L+2R перемешаны)
local function pick_side(bs)
    local mode = bs and bs.side_mode:get() or "Alternate"
    if not bs or bs.yaw:get() ~= "L/R" or mode == "Alternate" then return not eng.side end

    local new
    if mode == "Random" then
        new = math.random(100) <= bs.side_bias:get()
    else
        if #eng.sidebag == 0 then eng.sidebag = shuffle({true, true, false, false}) end
        new = table.remove(eng.sidebag)
    end

    if new == eng.side then eng.streak = eng.streak + 1 else eng.streak = 1 end
    if eng.streak > bs.side_streak:get() then
        new = not eng.side
        eng.streak = 1
    end
    return new
end

-- ----- Bruteforce: смена фаз (старая система; при удалённой вкладке остаётся выключенной) -----
local function bf_step()
    local tick = globals.tickcount
    if eng.phase > 1 and (tick >= eng.phase_until or eng.phase_until - tick > 64 * 30) then eng.phase = 1 end

    if not M.bf_on:get() then
        eng.phase = 1
        eng.bf_imp, eng.bf_hurt = false, false
        return
    end

    local imp, hurt = eng.bf_imp, eng.bf_hurt
    eng.bf_imp, eng.bf_hurt = false, false
    if not (imp or hurt) then return end

    local sel = M.bf_trigger:get()
    local fire = (hurt and has(sel, "Got hit")) or (imp and not hurt and has(sel, "Enemy miss"))
    if not fire then return end

    local cnt = M.bf_count:get()
    if eng.phase < 2 then eng.phase = 2
    elseif eng.phase < cnt then eng.phase = eng.phase + 1
    else eng.phase = 2 end

    local ti = safe(function() return globals.tickinterval end, 1 / 64)
    eng.phase_until = tick + math.floor(rint(M.bf_tmin:get(), M.bf_tmax:get()) / ti)
    eng.limL = nil
    if M.bf_flip:get() then eng.bf_flip = true end
end

-- какой набор углов сейчас работает: базовый Builder или фаза Bruteforce
local function active_block(bs)
    if bs and eng.phase > 1 and M.bf_on:get() and state_ok(M.bf_states, cur_raw) then
        return BF.ph[eng.phase] or bs
    end
    return bs
end

local function engine_tick(cmd, bs)
    local tick = globals.tickcount
    if tick < eng.last then eng.last, eng.blast, eng.fz_cd = tick, tick, 0 end
    if choked(cmd) ~= 0 then return end
    local blk = active_block(bs)

    -- Bruteforce: мгновенная смена стороны при срабатывании
    if eng.bf_flip then
        eng.bf_flip = false
        if blk then
            eng.side = not eng.side
            eng.flipped = true
            eng.streak = 1
            eng.bside = not eng.bside
            eng.last = tick
            eng.wait = math.max(1, side_wait(blk, eng.side))
            return
        end
    end

    if tick - eng.last < eng.wait then return end
    eng.last = tick
    eng.ron = math.random() < 0.5
    eng.rside = math.random() < 0.5

    -- Freeze (только из базового Builder): с шансом AA замирает на min..max тиков
    if bs and bs.freeze:get() and tick >= eng.fz_cd and math.random(100) <= bs.freeze_chance:get() then
        local dur = math.max(1, rint(bs.freeze_min:get(), bs.freeze_max:get()))
        local fo = bs.freeze_side:get()
        local want = eng.side
        if fo == "Left" then want = true
        elseif fo == "Right" then want = false
        elseif fo == "Random" then want = math.random() < 0.5 end
        if want ~= eng.side then eng.flipped = true end
        eng.side = want
        eng.wait = dur
        eng.frozen = true
        eng.fz_cd = tick + dur + bs.freeze_cd:get()
        return
    end

    eng.frozen = false
    local new = pick_side(blk)
    if new ~= eng.side then eng.flipped = true end
    eng.side = new
    eng.wait = math.max(1, side_wait(blk, eng.side))
end

-- ========== HEAD YAW ИЗ БИЛДЕРА (только yaw offset, body не трогает) ==========
local function compute_yaw(bs, tick)
    local yaw
    local left = eng.side
    if bs.yaw:get() == "L/R" then
        yaw = left and bs.yaw_left:get() or bs.yaw_right:get()
        if bs.rs:get() then
            -- случайная добавка 0..N поверх угла стороны (влево — минус, вправо — плюс)
            local r = left and bs.rs_left:get() or bs.rs_right:get()
            local add = math.random() * r
            yaw = yaw + (left and -add or add)
        end
    else
        yaw = bs.yaw_static:get()
    end

    local m = bs.mod:get()
    if m ~= "None" then
        local a = bs.mod_amount:get()
        if bs.mod_rand:get() then a = lerp(a, bs.mod_amount2:get(), math.random()) end
        if m == "Jitter" then
            yaw = yaw + ((tick % 2 == 0) and a / 2 or -a / 2)
        elseif m == "X-ways" then
            local n = math.max(3, bs.mod_ways:get())
            yaw = yaw + ((tick % n) / (n - 1) * 2 - 1) * a / 2
        elseif m == "Sway" then
            yaw = yaw + math.sin(globals.curtime * bs.mod_speed:get()) * a / 2
        elseif m == "Random" then
            yaw = yaw + (math.random() * 2 - 1) * a / 2
        elseif m == "Spin" then
            local rng = math.max(1, math.abs(a))
            yaw = yaw + ((globals.curtime * bs.mod_speed:get() * 45) % rng) - rng / 2
        end
    end

    -- шум угла: не повторяем предыдущий угол ближе чем на gap
    local noise = bs.yaw_noise:get()
    if noise > 0 then
        local gap, y = bs.yaw_gap:get(), yaw
        for _ = 1, 5 do
            y = yaw + (math.random() * 2 - 1) * noise
            if math.abs(norm_yaw(y - eng.last_yaw)) >= gap then break end
        end
        yaw = y
    end
    eng.last_yaw = yaw
    return norm_yaw(yaw)
end

local AAX = {}

-- true, если основная угроза слева от нашего взгляда; nil, если угрозы нет

end

-- ========== BODY YAW ==========
local function set_limits(l, r)
    ovr(R.left_limit, l)
    ovr(R.right_limit, r or l)
end

local function apply_body(cfg, cmd)
    if not cfg.body:get() then ovr(R.body_yaw, false) return end
    ovr(R.body_yaw, true)

    -- лимиты: пересчитываются только когда сработал мастер-таймер (или сменился набор настроек)
    local md = H.md
    local lim_l, lim_r
    if md.fire or md.lim_cfg ~= cfg or not md.lim_l then
        -- Static / Random (бросок при смене стороны) / Sway (плавно между min и max)
        local lm = cfg.body_limits:get()
        lim_l, lim_r = 0, 0
        if lm == "Static" then
            lim_l, lim_r = cfg.lim_l:get(), cfg.lim_r:get()
        elseif lm == "Random" then
            if eng.flipped or not eng.limL then
                eng.limL = rint(cfg.lim_min:get(), cfg.lim_max:get())
                eng.limR = rint(cfg.lim_min:get(), cfg.lim_max:get())
            end
            lim_l, lim_r = eng.limL, eng.limR
        else
            local t = (math.sin(globals.curtime * 2.2) + 1) / 2
            lim_l = lerp(cfg.lim_min:get(), cfg.lim_max:get(), t)
            lim_r = lerp(cfg.lim_min:get(), cfg.lim_max:get(), 1 - t)
        end
        local noise = cfg.lim_noise:get()
        if noise > 0 then
            lim_l = clamp(lim_l + (math.random() * 2 - 1) * noise, 0, 60)
            lim_r = clamp(lim_r + (math.random() * 2 - 1) * noise, 0, 60)
        end
        md.lim_cfg, md.lim_l, md.lim_r = cfg, lim_l, lim_r
    else
        lim_l, lim_r = md.lim_l, md.lim_r
    end
    set_limits(lim_l, lim_r)

    local opts = safe(function() return cfg.body_opts:get() end, {})
    if type(opts) ~= "table" then opts = {} end
    local jit = has(opts, "Jitter") or has(opts, "Randomize Jitter")
    ovr(R.body_options, opts)
    ovr(R.body_fs, cfg.body_fs:get())

    local state = cfg.body_state:get()
    if state == "Delay" then
        local tick = globals.tickcount
        if tick < eng.blast then eng.blast = tick end
        if choked(cmd) == 0 and tick - eng.blast >= eng.bwait then
            eng.bside = not eng.bside
            eng.blast = tick
            local dmin = cfg.body_delay:get()
            eng.bwait = rint(dmin, math.max(dmin, cfg.body_delay_max:get()))
        end
    end

    if not jit then
        local inv = cfg.body_inv:get()
        local sync = cfg.body_sync:get()
        if sync == "Follow head" then inv = (inv ~= eng.side)
        elseif sync == "Opposite head" then inv = (inv ~= (not eng.side))
        elseif sync == "Random" then inv = (inv ~= eng.rside)
        elseif sync == "Toward threat" or sync == "Away from threat" then
            local tl = AAX.threat_left()
            if tl ~= nil then
                if sync == "Away from threat" then tl = not tl end
                inv = (inv ~= tl)
            end
        end
        if state == "Delay" then inv = (inv ~= eng.bside)
        elseif state == "Random" then inv = (inv ~= eng.rside) end
        ovr(R.inverter, inv)
    end

    if state == "Random" and not eng.ron then ovr(R.body_yaw, false) end
end

-- ========== DEFENSIVE ИЗ ВКЛАДКИ DEFENSIVE ==========
-- шаг паттернов defensive: счётчик растёт раз в N тиков (delay), все Swap/Switch/3-way идут по нему
local dfs = {last = 0, wait = 1, step = 0}
local function def_step(ds)
    local tick = globals.tickcount
    if tick < dfs.last then dfs.last = tick end
    if tick - dfs.last < dfs.wait then return end
    dfs.last = tick
    dfs.step = dfs.step + 1
    local m, lo, hi = ds.dl_mode:get(), ds.dl_min:get(), ds.dl_max:get()
    dfs.wait = (m == "Static") and lo or pick_range(lo, hi, "D", m, true)
    dfs.wait = math.max(1, dfs.wait)
end

local function apply_defensive(ds, cmd)
    if not ds or manual_dir ~= 0 or not def.active then return false end
    def_step(ds)
    local tick = dfs.step

    local pitch = pattern(ds.pitch:get(), ds.p1:get(), ds.p2:get(), ds.pspd:get(), tick)
    local yaw   = pattern(ds.yaw:get(), ds.y1:get(), ds.y2:get(), ds.yspd:get(), tick)
    local m = ds.mod:get()
    if m ~= "None" then yaw = yaw + pattern(m, ds.m1:get(), ds.m2:get(), ds.mspd:get(), tick) end

    hidden_angles(clamp(pitch, -89, 89), norm_yaw(yaw))
    apply_body(ds, cmd)
    return true
end

-- ========== BRUTEFORCE: СОБЫТИЯ ==========
-- пуля пролетела рядом с головой (промах врага) — флаг; урон по нам — отдельный флаг.
-- Эти события кормят Anti-Bruteforce (фиксированная дистанция 70u).
on("bullet_impact", function(e)
    local lp = entity.get_local_player()
    if not lp or not lp:is_alive() or not M.enable:get() then return end

    local shooter = entity.get(e.userid, true)
    if not shooter or shooter == lp or not shooter:is_enemy() then return end

    local ok, d = pcall(function()
        local s, h = shooter:get_eye_position(), lp:get_eye_position()
        local dx, dy, dz = e.x - s.x, e.y - s.y, e.z - s.z
        local len2 = dx * dx + dy * dy + dz * dz
        if len2 < 1 then return 9999 end
        local t = clamp(((h.x - s.x) * dx + (h.y - s.y) * dy + (h.z - s.z) * dz) / len2, 0, 1)
        local px, py, pz = s.x + dx * t, s.y + dy * t, s.z + dz * t
        return math.sqrt((px - h.x) ^ 2 + (py - h.y) ^ 2 + (pz - h.z) ^ 2)
    end)
    if ok and d then
        if M.bf_on:get() and d <= M.bf_dist:get() then eng.bf_imp = true end
        if d <= 70 then H.bf.imp = true end
    end
end)

on("player_hurt", function(e)
    local lp = entity.get_local_player()
    if not lp or not M.enable:get() then return end
    if entity.get(e.userid, true) ~= lp then return end
    local attacker = entity.get(e.attacker, true)
    if not attacker or attacker == lp or not attacker:is_enemy() then return end

    if M.bf_on:get() then eng.bf_hurt = true end
    H.bf.hurt = true

    -- Anti-massive: повторные попадания за короткое окно = наш desync читают
    if M.h_antimf:get() then
        local now = globals.realtime
        local hs = H.am.hurts
        hs[#hs + 1] = now
        for i = #hs, 1, -1 do
            if now - hs[i] > 3 then table.remove(hs, i) end
        end
        if #hs >= 2 then H.am.untl = now + 2.5 end
    end
end)

-- ========== МОДУЛЬ: DEFENSIVE FLICK ==========
local function module_flick(cmd, base_yaw)
    local tick = globals.tickcount
    local mode, hold, interval = E.ff_mode:get(), E.ff_hold:get(), E.ff_interval:get()
    local cycle, flicking

    if mode == "Defensive" then
        cycle = def.windows
        flicking = def.active and def.age < hold
    else
        cycle = math.floor(tick / interval)
        flicking = (tick % interval) < hold
    end

    if cycle ~= ff.cycle then
        ff.cycle = cycle
        local d = E.ff_dir:get()
        if d == "Left" then ff.sign = -1
        elseif d == "Right" then ff.sign = 1
        elseif d == "Alternate" then ff.sign = -ff.sign
        else ff.sign = math.random(0, 1) == 0 and -1 or 1 end
    end
    if not flicking then return end

    local angle = E.ff_angle:get() * ff.sign
    ovr(R.modifier, "Disabled")
    ovr(R.yaw_offset, norm_yaw(base_yaw + angle))
    ovr(R.body_yaw, true); ovr(R.body_options, {}); set_limits(60)
    ovr(R.inverter, ff.sign < 0)

    local opt = E.ff_pitch:get()
    local pv = flick_pitch_value(opt ~= "Keep" and opt or "Off")
    if def.active then
        hidden_angles(pv, angle)
    else
        if opt == "Up" then ovr(R.pitch, "Fake Up")
        elseif opt == "Down" then ovr(R.pitch, "Fake Down")
        elseif opt == "Zero" then ovr(R.pitch, "Disabled") end
    end
end

-- ========== МОДУЛЬ: MASSIVE FAKE ==========
local function module_massive(cmd, base_yaw)
    local style, lim, tick = E.mf_style:get(), E.mf_limit:get(), globals.tickcount

    if tick < mf.last then mf.last = tick end
    if style ~= "Jitter max" and tick - mf.last >= E.mf_flip:get() and choked(cmd) == 0 then
        mf.side = -mf.side
        mf.last = tick
    end

    if style == "Jitter max" then
        ovr(R.modifier, "Center"); ovr(R.modifier_offset, 24)
        ovr(R.body_yaw, true); ovr(R.body_options, {"Jitter"}); set_limits(lim)
    else
        ovr(R.body_yaw, true); ovr(R.body_options, {}); set_limits(lim)
        ovr(R.inverter, mf.side < 0)
        ovr(R.modifier, "Disabled")
        ovr(R.yaw_offset, norm_yaw(base_yaw + mf.side * 8))
    end

    if style == "Defensive flick" and def.active and def.age < 3 then
        hidden_angles(flick_pitch_value(E.mf_pitch:get()), E.mf_angle:get() * mf.side)
    end
end

-- ========== МОДУЛЬ: EXTENDED YAW ==========
local function module_extended(cmd, base_yaw)
    local t = globals.curtime * E.ex_speed:get()
    local range, pat = E.ex_range:get(), E.ex_pattern:get()
    local sweep
    if pat == "Sine" then sweep = math.sin(t * 2)
    elseif pat == "Saw" then sweep = (t % 1) * 2 - 1
    elseif pat == "Zigzag" then sweep = math.abs((t % 2) - 1) * 2 - 1
    else sweep = math.random() * 2 - 1 end

    local noise = (math.random() * 2 - 1) * E.ex_noise:get()
    ovr(R.modifier, "Disabled")
    ovr(R.yaw_offset, norm_yaw(base_yaw + sweep * range + noise))

    ovr(R.body_yaw, true); ovr(R.body_options, {"Jitter"})
    if E.ex_vary:get() then
        local step = math.floor(globals.tickcount / 3)
        if step ~= ex.last then ex.last = step; ex.limit = math.random(34, 60) end
        set_limits(ex.limit)
    else
        set_limits(60)
    end
end

-- ========== МОДУЛЬ: ANTI-SKEET ==========
local SK_LIMITS = {29, 38, 45, 52, 58, 60}
local function module_antiskeet(cmd, base_yaw)
    local tick = globals.tickcount
    if tick < sk.last then sk.last = tick end

    if choked(cmd) == 0 and tick - sk.last >= sk.wait then
        sk.last = tick
        local lo = E.sk_min:get()
        sk.wait = rint(lo, math.max(lo, E.sk_max:get()))

        local nxt = math.random() < 0.5
        if nxt == sk.side then sk.run = sk.run + 1 else sk.run = 1 end
        if sk.run > 2 then nxt = not nxt; sk.run = 1 end
        sk.side = nxt
        sk.limit = E.sk_vary:get() and SK_LIMITS[math.random(1, #SK_LIMITS)] or 60
    end

    ovr(R.body_yaw, true); ovr(R.body_options, {})
    set_limits(sk.limit); ovr(R.inverter, sk.side)
    ovr(R.modifier, "Disabled")
    ovr(R.yaw_offset, norm_yaw(base_yaw + (sk.side and -1 or 1) * rint(0, E.sk_jit:get())))
end

-- ========== МОДУЛЬ: SCOLIOSIS ==========
local function module_scoliosis(cmd, base_yaw)
    local mode, amt, spd = E.sc_mode:get(), E.sc_amount:get(), E.sc_speed:get()
    local t = globals.tickcount
    local pitch, yaw

    if mode == "Twist" then
        local s = (t % 2 == 0) and 1 or -1
        pitch, yaw = s * amt, s * amt * 2
    elseif mode == "Wave" then
        pitch = math.sin(globals.curtime * spd) * amt
        yaw = math.cos(globals.curtime * spd) * amt * 2
    else
        pitch, yaw = (math.random() * 2 - 1) * amt, (math.random() * 2 - 1) * 180
    end

    if def.active then
        hidden_angles(clamp(pitch, -89, 89), norm_yaw(yaw))
    else
        ovr(R.yaw_offset, norm_yaw(base_yaw + yaw * 0.25))
    end
    if E.sc_roll:get() and R.roll then ovr(R.roll, clamp(pitch * 0.6, -50, 50)) end
end

-- ========== FREESTANDING (авто: у стены + виден враг) ==========
local function near_wall(lp)
    local ok, res = pcall(function()
        local o = lp:get_origin() + vector(0, 0, 40)
        local yaw = render.camera_angles().y
        local dist = M.fs_wall:get()
        for a = 0, 315, 45 do
            local r = math.rad(yaw + a)
            local e = o + vector(math.cos(r) * dist, math.sin(r) * dist, 0)
            local tr = utils.trace_line(o, e, lp)
            if tr and tr.fraction < 0.95 then return true end
        end
        return false
    end)
    if ok then return res end
    return nil
end

local function update_freestanding(lp, raw)
    local want = false
    if M.fstand:get() and manual_dir == 0 and not is_warmup() then
        local sel = M.fs_cond:get()
        local standing = has(sel, "Standing") and (raw == 2 or raw == 7)
        local moving = has(sel, "Moving") and (raw == 3 or raw == 4 or raw == 8)
        if (standing or moving) and threat() ~= nil then
            want = near_wall(lp) ~= false
        end
    end
    if want then fs.hold = globals.tickcount + 8 end
    if globals.tickcount < fs.hold - 64 then fs.hold = 0 end
    fs.active = globals.tickcount < fs.hold
end

-- ========== AIR LAG ==========
-- Лагаем только в состоянии In Air (raw == 5). In Air + C и земля не затрагиваются.
-- Ideal tick: чанкаем ровно столько тиков, чтобы смещение между отправками превысило 64 юнита
-- (так ломается lag compensation), чем быстрее движение — тем меньше тиков нужно.
local function apply_air_lag(cmd, lp, raw)
    local function reset() ovr(R.fl_limit); ovr(R.fl_var); air.choke = 0; air.target = 0 end

    if not E.al:get() or raw ~= 5 then reset() return false end

    local v = lp.m_vecVelocity
    local trig = E.al_trigger:get()
    local go = trig == "Always in air"
        or (trig == "After jump start" and air.ticks >= 3)
        or (trig == "At peak" and math.abs(v.z) <= 90)
    if not go then reset() return false end

    local mode, maxt = E.al_mode:get(), E.al_limit:get()
    local ticks
    if mode == "Ideal tick" then
        local speed = math.sqrt(v.x * v.x + v.y * v.y + v.z * v.z)
        local ti = safe(function() return globals.tickinterval end, 1 / 64)
        if speed > 1 then
            ticks = math.ceil(64 / (speed * ti)) + E.al_extra:get()
        else
            ticks = maxt
        end
    elseif mode == "Random" then
        if air.choke == 0 or air.target == 0 then air.target = rint(2, maxt) end
        ticks = air.target
    else
        ticks = maxt
    end
    ticks = clamp(ticks, 2, maxt)

    ovr(R.fl_enable, true); ovr(R.fl_limit, ticks); ovr(R.fl_var, 0)

    if air.choke < ticks then
        pcall(function() cmd.send_packet = false end)
        air.choke = air.choke + 1
    else
        pcall(function() cmd.send_packet = true end)
        air.choke = 0
    end
    return E.al_def:get()
end

-- ========== AIR STOP ==========
local function weapon_ready(lp)
    local w = lp:get_player_weapon()
    if not w then return false end
    return safe(function()
        local now = globals.curtime
        return w.m_flNextPrimaryAttack <= now and lp.m_flNextAttack <= now
    end, true)
end

local function apply_air_stop(cmd, lp, raw)
    if not E.as:get() or not state_ok(E.as_states, raw) then return end
    if E.as_target:get() and not threat() then return end
    if E.as_ready:get() and not weapon_ready(lp) then return end

    local vel = lp.m_vecVelocity
    local speed = vel:length2d()
    if speed < E.as_speed:get() then return end

    if E.as_mode:get() == "Zero input" then
        cmd.forwardmove, cmd.sidemove = 0, 0
        pcall(function()
            cmd.in_forward, cmd.in_back, cmd.in_moveleft, cmd.in_moveright = false, false, false, false
        end)
        return
    end

    -- counter-strafe: тянем в сторону, противоположную скорости (в системе координат взгляда)
    local yaw = math.rad(cmd.view_angles.y)
    local fwd  = vel.x * math.cos(yaw) + vel.y * math.sin(yaw)
    local side = vel.x * math.sin(yaw) - vel.y * math.cos(yaw)
    local mag = math.sqrt(fwd * fwd + side * side)
    if mag < 1 then return end
    local k = clamp(mag / 25, 0, 1) * 450
    cmd.forwardmove = clamp(-fwd / mag * k, -450, 450)
    cmd.sidemove    = clamp(-side / mag * k, -450, 450)
end

-- встроенный резольвер NL как «базовый»; наш RES работает и в режиме Assist
function H.res_on() return M.res:get() or M.h_assist:get() end

-- Strict Hitboxes Only: сканируем только Head / Chest / Stomach (руки, ноги, стопы отключены)
function H.apply_hitboxes()
    local want = M.enable:get() and M.h_strict:get()
    local tick = globals.tickcount
    if tick < H.hb_last then H.hb_last = 0 end
    if want ~= H.hb_on or (want and tick - H.hb_last > 64) then
        H.hb_last, H.hb_on = tick, want
        for i = 1, #H.hb do
            if want then ovr(H.hb[i], {"Head", "Chest", "Stomach"}) else ovr(H.hb[i]) end
        end
    end
end

-- ЕДИНЫЙ мастер-таймер: все пересчёты AA идут только когда он сработал
function H.master(cmd)
    local md = H.md
    local ms_ = M.h_delay:get()
    local tick = globals.tickcount
    if tick < md.last then md.last = 0 end
    if ms_ <= 0 then
        md.fire, md.last = true, tick
        return
    end
    local need = math.max(1, math.ceil(ms_ / (H.ti * 1000)))
    if choked(cmd) == 0 and tick - md.last >= need then
        md.last, md.fire = tick, true
    else
        md.fire = false
    end
end

-- Anti-Bruteforce: 1-й промах -> фаза 1, 2-й -> фаза 2 ... по кругу до 5; сброс при нашем попадании
function H.bf_step()
    local b = H.bf
    if not M.h_bfon:get() then b.idx, b.miss, b.imp, b.hurt = 0, 0, false, false return end
    local imp, hurt = b.imp, b.hurt
    b.imp, b.hurt = false, false
    if imp and not hurt then
        b.miss = b.miss + 1
        b.idx = ((b.miss - 1) % 5) + 1
        b.t = globals.realtime
        eng.bf_flip = true
        H.log_push("hit", string.format("enemy miss #%d  forced phase %d (%+d°)", b.miss, b.idx,
            math.floor(M.h_bf[b.idx]:get())))
    end
end

-- Body Yaw Modifier + Bruteforce offset + Anti-Massive Fake
function H.apply_hyper(cmd, lp, base_yaw)
    if manual_dir ~= 0 then return end
    local md, am, b = H.md, H.am, H.bf
    local mode = M.h_mod:get()
    local tick, now = globals.tickcount, globals.realtime

    if md.fire then
        if mode ~= "Disabled" then
            local range = M.h_range:get()
            local cn = cmd.command_number or tick
            if mode == "Fixed Jitter" then
                -- аппаратная инверсия +range / -range на каждый запуск таймера (не залипает при любом delay)
                md.jpos = not md.jpos
                md.jit = md.jpos and range or -range
                md.jside = md.jpos
            elseif mode == "True Random Jitter" then
                -- никогда около нуля: модуль в диапазоне [35%..100%] от range
                local lo = math.max(1, math.floor(range * 0.35))
                local mag = math.random(lo, math.max(lo, math.floor(range)))
                md.jpos = math.random() < 0.5
                md.jit = md.jpos and mag or -mag
                md.jside = md.jpos
            else
                -- Micro-Sway: синусоида по номеру команды
                local s = math.sin(cn * 0.15)
                md.jit = s * range
                md.jside = s >= 0
            end
        end

        -- Anti-Massive Fake: следим за desync и за повторными промахами/попаданиями
        if M.h_antimf:get() then
            local lby = lp.m_flLowerBodyYawTarget
            if lby then
                local delta = math.abs(norm_yaw(cmd.view_angles.y - lby))
                if delta > 62 and delta < 178 then am.untl = math.max(am.untl, now + 1.2) end
            end
            if b.miss >= 2 and now - b.t < 3 then am.untl = math.max(am.untl, now + 1.5) end
            if tick < am.last then am.last = 0 end
            if now < am.untl and tick - am.last >= am.wait then
                am.last = tick
                am.wait = rint(2, 4)
                am.side = not am.side
            end
        end
    end

    local massive = M.h_antimf:get() and now < am.untl
    local bfo = (b.idx > 0 and M.h_bfon:get()) and M.h_bf[b.idx]:get() or 0

    if mode ~= "Disabled" or bfo ~= 0 then
        local yaw = base_yaw
        if mode ~= "Disabled" then yaw = yaw + md.jit end
        yaw = yaw + bfo
        ovr(R.yaw_offset, norm_yaw(yaw))
    end
    if mode ~= "Disabled" or massive then
        ovr(R.body_yaw, true); ovr(R.body_options, {})
        ovr(R.inverter, massive and am.side or md.jside)
    end
end

-- ========== ГЛАВНАЯ ФУНКЦИЯ АНТИ-АИМА ==========
local function apply_antiaim(cmd, lp)
    track_defensive(lp)
    update_manual()

    local raw, on_ground = get_state(cmd, lp)
    cur_raw = raw
    if on_ground then air.ticks = 0 else air.ticks = air.ticks + 1 end

    local team = lp.m_iTeamNum
    local bs = resolve(Bs, team, raw)
    local ds = resolve(Ds, team, raw)
    local tick = globals.tickcount
    cur_bs = bs

    clear_hidden()
    H.master(cmd)
    if H.md.fire then
        bf_step()
        H.bf_step()
        engine_tick(cmd, bs)
    end
    update_freestanding(lp, raw)
    local blk = active_block(bs)

    -- ----- основной AA из Builder -----
    local base_yaw = 0
    if bs then
        ovr(R.pitch, "Down"); ovr(R.yaw, "Backward")
        ovr(R.yaw_base, M.yaw_base:get()); ovr(R.freestanding, false)
        ovr(R.modifier, "Disabled")
        if H.md.fire or not H.md.yaw then H.md.yaw = compute_yaw(blk, tick) end
        base_yaw = H.md.yaw
        ovr(R.yaw_offset, norm_yaw(manual_dir ~= 0 and manual_dir or base_yaw))
        apply_body(blk, cmd)
        H.apply_hyper(cmd, lp, base_yaw)
    end

    -- ----- defensive из вкладки Defensive -----
    apply_defensive(ds, cmd)

    -- ----- эксплойты (последний побеждает) -----
    local need_def = ds ~= nil
    if manual_dir == 0 then
        if E.ex:get() and state_ok(E.ex_states, raw) then module_extended(cmd, base_yaw) end
        if E.sk:get() and state_ok(E.sk_states, raw) then module_antiskeet(cmd, base_yaw) end
        if E.mf:get() and state_ok(E.mf_states, raw) then
            module_massive(cmd, base_yaw)
            if E.mf_style:get() == "Defensive flick" then need_def = true end
        end
        if E.sc:get() and state_ok(E.sc_states, raw) then module_scoliosis(cmd, base_yaw); need_def = true end
        if E.ff:get() and state_ok(E.ff_states, raw) then
            module_flick(cmd, base_yaw)
            if E.ff_mode:get() == "Defensive" then need_def = true end
        end
    end

    -- ----- air lag / форсирование defensive -----
    local air_force = apply_air_lag(cmd, lp, raw)
    set_defensive(cmd, need_def or air_force)
    apply_fakeflick(cmd, raw)

    -- ----- Random Fakelag (пока air lag не занят) -----
    local air_busy = E.al:get() and raw == 5
    if M.rfl:get() and not air_busy then
        if choked(cmd) == 0 then
            rfl.limit = rint(M.rfl_min:get(), M.rfl_max:get())
        end
        ovr(R.fl_enable, true); ovr(R.fl_limit, rfl.limit)
    elseif not air_busy then
        ovr(R.fl_limit)
    end

    apply_specials(lp, raw)

    -- Legit AA on use: пока зажата E и рядом нет бомбы/заложника — смотрим назад, E не нажимается
    if M.legit_use:get() and cmd.in_use and not AAX.near_objective(lp) then
        overwrite("Legit")
        pcall(function() cmd.in_use = false end)
    end

    apply_air_stop(cmd, lp, raw)
    apply_air_hitchance(raw)

    if R.edge then
        ovr(R.edge, (M.edge:get() and R.fakeduck and R.fakeduck:get()) or false)
    end

    eng.flipped = false
end

local function fast_ladder(cmd, lp)
    if lp.m_MoveType ~= 9 then return end
    local pitch = render.camera_angles().x
    local va = cmd.view_angles
    va.y = math.floor(va.y + 0.5)

    if cmd.forwardmove == 0 and cmd.sidemove ~= 0 then
        va.x = 89; va.y = va.y + 180
        cmd.in_moveleft = cmd.sidemove > 0
        cmd.in_moveright = cmd.sidemove < 0
    end
    if cmd.forwardmove > 0 and pitch < 45 then
        va.x = 89
        cmd.in_moveright, cmd.in_moveleft = true, false
        cmd.in_forward, cmd.in_back = false, true
        va.y = va.y + (cmd.sidemove == 0 and 90 or (cmd.sidemove < 0 and 150 or 30))
    end
    if cmd.forwardmove < 0 then
        va.x = 89
        cmd.in_moveleft, cmd.in_moveright = true, false
        cmd.in_forward, cmd.in_back = true, false
        va.y = va.y + (cmd.sidemove == 0 and 90 or (cmd.sidemove > 0 and 150 or 30))
    end
end

-- row = { {текст, роль, [цвет]}, ... }; роли: tag (цвет), name, txt, dim
function H.draw_row(row, scrx, y, a)
    local _, txt = theme()
    local gap, total = 8, 0
    local ws = {}
    for i = 1, #row do
        local font = (row[i][2] == "tag" or row[i][2] == "name") and H.fb or H.fs
        ws[i] = text_width(row[i][1], font)
        total = total + ws[i]
    end
    total = total + gap * (#row - 1)
    local x = math.floor(scrx / 2 - total / 2)
    for i = 1, #row do
        local seg = row[i]
        local role = seg[2]
        local font = (role == "tag" or role == "name") and H.fb or H.fs
        local col
        if role == "tag" then col = seg[3] or txt
        elseif role == "name" then col = txt
        elseif role == "dim" then col = color(txt.r, txt.g, txt.b, 150)
        else col = color(txt.r, txt.g, txt.b, 220) end

        render.text(font, vector(x + 1, y + 1), color(0, 0, 0, math.floor(190 * a)), "", text)
        render.text(font, vector(x, y), color(col.r, col.g, col.b, math.floor(col.a * a)), "", text)
        x = x + ws[i] + gap
    end
end

-- ========== RESOLVER ==========
-- Что делает:
--  1) определяет сторону desync: по бегу / LBY на месте / по фазе джиттера eye yaw;
--  2) ловит LBY flick (на месте eye yaw и LBY расходятся → в момент обновления LBY берём полный угол);
--  3) масштабирует амплитуду по скорости и приседанию, как это делает animstate (адаптивная амплитуда);
--  4) помнит, какой угол сработал, отдельно для «стоит/идёт/в воздухе» и для каждой фазы джиттера;
--  5) после промахов не повторяет уже провальные углы, а идёт по списку кандидатов.
-- Результат пишется в body-yaw pose parameter (11).
-- Работает при включённом «Resolver» или «Assist Base Resolver».
local RES = {data = {}, miss_reasons = {resolver = true, correction = true, unknown = true}}

local function res_key(c) return string.format("%.2f", c) end

function RES.get(pl)
    local id = safe(function() return pl:get_index() end, nil)
    if not id then return nil end
    local d = RES.data[id]
    if not d then
        d = {miss = 0, side = 1, mult = 0, eye = 0, lby = 0, deltas = {}, eyes = {}, jit = false, phase = 1,
             center = 0, bad = {}, mem = {}, flick_until = 0, flick_mult = 0, key = "stand", amp = 1, streak = 0,
             score = {}, flip_tick = 0, intervals = {}, period = nil, last_phase = 1, pphase = 1}
        RES.data[id] = d
    end
    return d
end

-- множитель максимального угла по скорости и приседанию (как в animstate CS:GO)
function RES.amp_scale(pl, speed)
    local frac = clamp(speed / 260, 0, 1)
    local m = 1 - 0.5 * frac
    local duck = safe(function() return pl.m_flDuckAmount end, 0)
    if duck > 0 then m = m + duck * clamp(speed / 82, 0, 1) * (0.5 - m) end
    return clamp(m, 0.35, 1)
end

-- список кандидатов от самого вероятного к менее вероятному
function RES.candidates(guess, aggressive)
    if aggressive then
        return {guess, -guess, guess * 0.6, -guess * 0.6, 0, guess * 0.3, -guess * 0.3}
    end
    return {guess, -guess, guess * 0.5, -guess * 0.5, 0, guess * 0.75, -guess * 0.75, guess * 0.25, -guess * 0.25}
end

-- выбираем кандидата с лучшим накопленным счётом (попадание +2, промах −1), уже проваленные пропускаем
function RES.pick(d, guess, aggressive, mkey)
    local sc = d.score[mkey or ""] or {}
    local best, best_v = nil, -1000000000
    for i, c in ipairs(RES.candidates(guess, aggressive)) do
        local k = res_key(c)
        if not d.bad[k] then
            local v = (sc[k] or 0) - i * 0.01
            if v > best_v then best, best_v = c, v end
        end
    end
    if best then return best end
    d.bad = {}
    return guess
end

function RES.compute(pl, d)
    local eye = safe(function() return pl.m_angEyeAngles.y end, nil)
    if not eye then return end
    local tick = globals.tickcount

    -- ----- история eye yaw: джиттер, центр, фаза -----
    local delta = norm_yaw(eye - d.eye)
    d.eye = eye
    table.insert(d.deltas, delta)
    if #d.deltas > 8 then table.remove(d.deltas, 1) end
    table.insert(d.eyes, eye)
    if #d.eyes > 8 then table.remove(d.eyes, 1) end

    local flips = 0
    for i = 2, #d.deltas do
        local a, b = d.deltas[i - 1], d.deltas[i]
        if math.abs(a) > 12 and math.abs(b) > 12 and a * b < 0 then flips = flips + 1 end
    end
    d.jit = M.res_jit:get() and flips >= 3

    -- центр джиттера: среднее смещений относительно первого значения (без проблем на переходе ±180)
    local base, acc = d.eyes[1], 0
    for i = 1, #d.eyes do acc = acc + norm_yaw(d.eyes[i] - base) end
    d.center = norm_yaw(base + acc / #d.eyes)
    d.phase = (norm_yaw(eye - d.center) >= 0) and 1 or -1

    -- период джиттера: считаем тики между сменами фазы и предсказываем фазу на следующий тик
    if d.phase ~= d.last_phase then
        local iv = tick - d.flip_tick
        if d.flip_tick > 0 and iv > 0 and iv < 40 then
            table.insert(d.intervals, iv)
            if #d.intervals > 5 then table.remove(d.intervals, 1) end
            local sum = 0
            for _, v in ipairs(d.intervals) do sum = sum + v end
            d.period = sum / #d.intervals
        end
        d.flip_tick, d.last_phase = tick, d.phase
    end
    local ph = d.phase
    if d.jit and d.period and (tick - d.flip_tick + 1) >= d.period then ph = -ph end
    d.pphase = ph

    -- ----- скорость, состояние, сторона -----
    local vel = safe(function() return pl.m_vecVelocity end, nil)
    local speed = vel and vel:length2d() or 0
    local flags = safe(function() return pl.m_fFlags end, 1)
    local on_ground = bit.band(flags, 1) == 1

    if not on_ground then d.key = "air"
    elseif speed > 20 then d.key = "move"
    else d.key = "stand" end

    local lby = safe(function() return pl.m_flLowerBodyYawTarget end, eye)
    local lby_changed = math.abs(norm_yaw(lby - d.lby)) > 1
    d.lby = lby

    if speed > 20 and vel then
        local move_yaw = math.deg(math.atan2(vel.y, vel.x))
        local diff = norm_yaw(eye - move_yaw)
        if math.abs(diff) > 3 then d.side = diff >= 0 and 1 or -1 end
    else
        local diff = norm_yaw(eye - lby)
        if math.abs(diff) > 5 then d.side = diff >= 0 and 1 or -1 end
    end

    -- ----- LBY flick: обновление LBY на месте → реальный угол = LBY -----
    if M.res_flick:get() and d.key == "stand" and lby_changed then
        local diff = norm_yaw(eye - lby)
        if math.abs(diff) > 15 then
            d.flick_until = tick + 2
            d.flick_mult = (diff >= 0 and 1 or -1)
        end
    end

    -- ----- итоговый угол -----
    local mode = M.res_mode:get()
    local aggressive = mode == "Aggressive"
    -- на джиттере сторона привязана к фазе eye yaw
    local guess = d.side
    if d.jit then guess = d.pphase * (d.side >= 0 and 1 or -1) end

    local mkey = d.key .. (d.jit and (d.pphase > 0 and ":p" or ":n") or "")
    local mult
    if mode ~= "Brute" and tick < d.flick_until then
        mult = d.flick_mult
    elseif mode ~= "Brute" and M.res_mem:get() and d.mem[mkey] ~= nil and d.streak < (aggressive and 1 or 2) then
        mult = d.mem[mkey]
    elseif mode == "Brute" then
        mult = RES.pick(d, 1, false, mkey)
    else
        mult = RES.pick(d, guess, aggressive, mkey)
    end

    d.mult = mult
    d.mkey = mkey
    d.amp = M.res_amp:get() and RES.amp_scale(pl, speed) or 1
end

function RES.apply(pl, d)
    pcall(function()
        pl.m_flPoseParameter[11] = clamp(0.5 + d.mult * d.amp * M.res_max:get() / 116, 0, 1)
    end)
end

function RES.run(compute)
    -- встроенный резольвер Neverlose мешает: пока наш включён, глушим его (если пункт найден)
    -- в режиме Assist Base Resolver встроенный НЕ отключается
    local want_off = M.enable:get() and M.res:get() and M.res_native:get()
    if R.native_res then
        if want_off then ovr(R.native_res, false); RES.native_ovr = true
        elseif RES.native_ovr then ovr(R.native_res); RES.native_ovr = false end
    end

    if not (M.enable:get() and H.res_on()) then return end
    local lp = entity.get_local_player()
    if not lp or not lp:is_alive() then return end
    for _, pl in ipairs(safe(function() return entity.get_players(true) end, {})) do
        if safe(function() return pl:is_alive() end, false) and not safe(function() return pl:is_dormant() end, true) then
            local d = RES.get(pl)
            if d then
                if compute then RES.compute(pl, d) end
                RES.apply(pl, d)
            end
        end
    end
end

-- снимок на момент выстрела (мы запоминаем, чем именно стреляли)
function RES.snap(target)
    if not (H.res_on() and target) then return nil end
    local d = RES.get(target)
    if not d then return nil end
    return {mult = d.mult, key = d.mkey}
end

function RES.feedback(target, hit, reason, shot)
    if not (H.res_on() and target) then return end
    local d = RES.get(target)
    if not d then return end
    local used = shot and shot.mult or d.mult
    local key = shot and shot.key or d.mkey

    local sc = d.score[key or ""]
    if not sc then sc = {}; d.score[key or ""] = sc end
    local uk = res_key(used)

    if hit then
        d.miss, d.streak, d.bad = 0, 0, {}
        if key then d.mem[key] = used end
        sc[uk] = clamp((sc[uk] or 0) + 2, -5, 12)
    elseif RES.miss_reasons[reason] then
        d.miss, d.streak = d.miss + 1, d.streak + 1
        d.bad[uk] = true
        sc[uk] = clamp((sc[uk] or 0) - 1, -5, 12)
        if key and d.mem[key] == used then d.mem[key] = nil end
    end
end

function RES.info(target)
    if not (H.res_on() and M.res_log:get() and target) then return "" end
    local d = RES.get(target)
    if not d then return "" end
    return string.format("  ·  res %+d° m%d %s%s", math.floor(d.mult * d.amp * M.res_max:get()), d.miss, d.key or "",
        d.jit and " jit" or "")
end

on("net_update_end", function() RES.run(true) end)

local function console_log(parts)
    local line = ""
    for _, p in ipairs(parts) do line = line .. esc(p[1]) .. p[2] end
    if not pcall(print_raw, line) then

        for _, p in ipairs(parts) do plain = plain .. p[2] end

    end
end

on("aim_ack", function(e)
    local lp = entity.get_local_player()
    local target = e.target
    local logo, txt = theme()
    local out = M.logs:get() and M.log_out:get() or {}

    local name = safe(function() return target:get_name() end, "unknown")
    local hc   = math.floor(safe(function() return e.hitchance end, 0) + 0.5)
    local bt   = safe(function() return e.backtrack end, 0)
    local wdmg = safe(function() return e.wanted_damage end, 0)
    local whg  = HITBOXES[safe(function() return e.wanted_hitgroup end, 0)] or "?"
    local safety = safe(function() return e.safety end, 0)
    local dim = GRAY
    local detail = "hc " .. hc .. "%  ·  bt " .. tostring(bt) .. "t  ·  safe " .. tostring(safety) .. RES.info(target)
    local hcbt = "HC " .. hc .. "%  BT " .. tostring(bt) .. "t"

    if e.state == nil then
        stats.hit = stats.hit + 1
        RES.feedback(target, true, nil, fired[e.id] and fired[e.id].res)
        local hg    = HITBOXES[safe(function() return e.hitgroup end, 0)] or "?"
        local dmg   = safe(function() return e.damage end, 0)
        local hp    = math.max(0, safe(function() return target.m_iHealth end, 0))
        local alive = hp > 0

        -- Hyper: лог с фазой и сброс индекса bruteforce при нашем попадании
        H.log_push("hit", string.format("%s %s | %s | %s dmg | phase %d", alive and "hit" or "killed", name, hg,
            tostring(math.floor(dmg)), H.bf.idx))
        H.bf.idx, H.bf.miss = 0, 0

        -- строка в стиле скрина
        local mismatch = hg ~= whg
        local tag, tcol = alive and "HIT" or "KILL", H.GREEN
        if mismatch then tag, tcol = "MISMATCH", H.ORANGE end
        local row = {
            {tag, "tag", tcol}, {name, "name"},
            {math.floor(dmg) .. " dmg / " .. math.floor(hp) .. " hp", "txt"},
            {hg, "txt"}, {hcbt, "dim"},
        }
        if mismatch then row[#row + 1] = {"expected " .. math.floor(wdmg) .. " in " .. whg, "dim"} end

        if has(out, "On Screen") then
            local d2 = detail
            if hg ~= whg then d2 = "aimed " .. whg .. "  ·  " .. detail end
            notify("hit", {
                title = alive and "HIT" or "KILL", name = name, hitbox = hg,
                badge = (alive and "" or "KILL ") .. "-" .. dmg, detail = d2, row = row,
            })
        end
        if has(out, "In Console") then
            local parts = {
                {logo, "lethe"}, {dim, "  »  "}, {logo, alive and "hit " or "killed "}, {txt, name},
                {dim, " · "}, {logo, hg},
            }
            if hg ~= whg then parts[#parts + 1] = {dim, " (aimed " .. whg .. ")"} end
            parts[#parts + 1] = {dim, " · "}; parts[#parts + 1] = {logo, tostring(dmg)}; parts[#parts + 1] = {txt, " dmg"}
            if dmg ~= wdmg then parts[#parts + 1] = {dim, " (wanted " .. tostring(wdmg) .. ")"} end
            parts[#parts + 1] = {dim, " · " .. detail:gsub("  ·  ", " · ")}
            console_log(parts)
        end
    else
        local reason = tostring(e.state)
        if reason ~= "death" then stats.miss = stats.miss + 1 end

        if reason == "?" then
            reason = "unknown"
            local f = fired[e.id]
            if f and f.total_hits ~= safe(function() return lp.m_totalHitsOnServer end, f.total_hits) then
                reason = "damage rejection"
            end
        end

        RES.feedback(target, false, reason, fired[e.id] and fired[e.id].res)
        H.log_push("miss", string.format("missed %s | %s | %s | phase %d", name, whg, reason, H.bf.idx))

        if has(out, "On Screen") then
            notify("miss", {
                title = "MISS", name = name, hitbox = whg, badge = reason, detail = detail,
                row = {{"MISS", "tag", H.MISSC}, {name, "name"}, {reason, "txt"}, {"hitbox: " .. whg, "dim"}, {hcbt, "dim"}},
            })
        end
        if has(out, "In Console") then
            console_log({
                {logo, "lethe"}, {dim, "  »  "}, {RED, "missed "}, {txt, name}, {dim, " · "}, {logo, whg},
                {dim, " · reason: "}, {RED, reason},
                {dim, " · " .. detail:gsub("  ·  ", " · ")},
            })
        end
    end
end)

-- ----- простые иконки из примитивов (rect / rect_outline / line) -----
function H.icon(kind, x, cy, col)
    local function r(x1, y1, x2, y2, rd, outline)
        local p1, p2 = vector(x + x1, cy + y1), vector(x + x2, cy + y2)
        if outline then pcall(render.rect_outline, p1, p2, col, 1, rd or 0)
        else render.rect(p1, p2, col, rd or 0) end
    end
    local function ln(x1, y1, x2, y2)
        pcall(render.line, vector(x + x1, cy + y1), vector(x + x2, cy + y2), col)
    end
    if kind == "user" then r(4, -6, 8, -2, 2); r(1, -1, 11, 5, 3)
    elseif kind == "fps" then r(0, -5, 12, 3, 2, true); r(4, 4, 8, 6, 1)
    elseif kind == "ping" then r(0, 2, 2, 5); r(3, 0, 5, 5); r(6, -3, 8, 5); r(9, -6, 11, 5)
    elseif kind == "time" then r(0, -6, 12, 6, 6, true); ln(6, 0, 6, -3); ln(6, 0, 9, 0)
    elseif kind == "da" then r(0, -6, 12, 6, 6, true); r(5, -1, 7, 1)
    elseif kind == "yaw" then ln(2, -5, 8, 0); ln(8, 0, 2, 5); ln(3, -5, 9, 0); ln(9, 0, 3, 5)
    elseif kind == "hit" then r(0, 0, 3, 5); r(4, -3, 7, 5); r(8, -6, 11, 5) end
end

-- ping: пробуем несколько источников, обновляем раз в 0.5 сек
function H.get_ping()
    local now = globals.realtime
    if now - H.ping_t < 0.5 then return H.ping end
    H.ping_t = now
    local lp = entity.get_local_player()
    local p = lp and safe(function() return entity.get_player_resource().m_iPing[lp:get_index()] end, nil)
    if not p then
        p = safe(function()
            local l = utils.net_channel().latency
            local v = l[1] or l[0]
            return v < 1 and v * 1000 or v
        end, nil)
    end
    if p then H.ping = math.floor(p + 0.5) end
    return H.ping
end

local function draw_watermark()
    local scr = render.screen_size()
    local logo, txt = theme()
    local items = M.wm_items:get()

    local segs = {}
    if has(items, "Script name") then segs[#segs + 1] = {kind = "name", text = gradient_text("lethe", logo, color(255, 255, 255, 255), 1)} end
    if has(items, "Username") then segs[#segs + 1] = {kind = "user", text = username} end
    if has(items, "FPS") then
        segs[#segs + 1] = {kind = "fps", text = math.floor(1 / math.max(globals.frametime, 0.001) + 0.5) .. " fps"}
    end
    if has(items, "Ping") then segs[#segs + 1] = {kind = "ping", text = H.get_ping() .. " ms"} end
    if has(items, "Time") then segs[#segs + 1] = {kind = "time", text = safe(function() return os.date("%H:%M") end, "--:--")} end
    if #segs == 0 then return end

    local pad, gap, ico, h = 9, 14, 14, 24
    local w = pad * 2 + gap * (#segs - 1)
    for _, s in ipairs(segs) do
        s.w = text_width(s.text) + (s.kind ~= "name" and (ico + 4) or 0)
        w = w + s.w
    end

    local pos = M.wm_type:get()
    local x, y
    if pos == "Top right" then x, y = scr.x - w - 16, 14
    elseif pos == "Top left" then x, y = 16, 14
    elseif pos == "Left" then x, y = 20, scr.y / 2
    elseif pos == "Right" then x, y = scr.x - 20 - w, scr.y / 2
    else x, y = scr.x / 2 - w / 2, scr.y - 46 end

    panel(x, y, w, h, 1, nil, 8)
    local cx = x + pad
    local ic = color(logo.r, logo.g, logo.b, 255)
    for _, s in ipairs(segs) do
        if s.kind ~= "name" then
            H.icon(s.kind, cx, y + h / 2, ic)
            render.text(1, vector(cx + ico + 4, y + 5), color(txt.r, txt.g, txt.b, 255), "", s.text)
        else
            render.text(1, vector(cx, y + 5), color(255, 255, 255, 255), "", s.text)
        end
        cx = cx + s.w + gap
    end
end

function V.cut(s, n)
    if #s <= n then return s end
    local c = n
    while c > 0 do
        local b = s:byte(c + 1)
        if b and b >= 128 and b < 192 then c = c - 1 else break end
    end
    return s:sub(1, c) .. "..."
end

                render.rect(vector(px, py), vector(px + cell, py + cell), col, 0)
            end
        end
    end
end

-- дуга под плашкой здоровья: идёт по нижним закруглениям и прямой между ними
function V.arc_points(x, y, w, h, off)
    local r, pts = h / 2, {}
    local function arc(cx, cy, a0, a1)
        for i = 0, 8 do
            local a = math.rad(lerp(a0, a1, i / 8))
            pts[#pts + 1] = {cx + math.cos(a) * (r + off), cy + math.sin(a) * (r + off)}
        end
    end
    arc(x + r, y + r, 165, 90)
    arc(x + w - r, y + r, 90, 15)
    return pts
end

function V.draw_hud(lp, sh)
    local hp = math.max(0, safe(function() return lp.m_iHealth end, 0))
    local ar = math.max(0, safe(function() return lp.m_ArmorValue end, 0))
    local money = safe(function() return lp.m_iAccount end, 0)
    local x, hy = 24, sh.y - 64
    local my = hy - 36

    -- деньги
    local mt = "$" .. money
    V.glass(x, my, 14 + 16 + 8 + text_width(mt, V.fb) + 16, 28, 1)
    V.pix(x + 14, my + 8, 2, V.ICON.wallet, V.BLUE)
    V.tx(V.fb, x + 14 + 16 + 8, my + 14, V.WHITE, mt)

    -- здоровье + дуга снизу
    local ht = tostring(hp)
    local hw = 16 + 21 + 10 + text_width(ht, V.fbig) + 18
    V.glass(x, hy, hw, 42, 1)
    V.pix(x + 16, hy + 12, 3, V.ICON.heart, V.BLUE)
    V.tx(V.fbig, x + 16 + 21 + 10, hy + 21, V.WHITE, ht)
    local pts = V.arc_points(x, hy, hw, 42, 3)
    local n = #pts - 1
    local k = math.floor(n * clamp(hp / 100, 0, 1) + 0.5)
    for i = 1, k do
        V.ln(pts[i][1], pts[i][2], pts[i + 1][1], pts[i + 1][2], V.BLUE)
        V.ln(pts[i][1], pts[i][2] + 1, pts[i + 1][1], pts[i + 1][2] + 1, V.BLUE)
    end

    -- броня
    local at = tostring(ar)
    local aw = 16 + 21 + 10 + text_width(at, V.fbig) + 18
    local ax = x + hw + 10
    V.glass(ax, hy, aw, 42, 1)
    V.pix(ax + 16, hy + 10, 3, V.ICON.shield, V.BLUE)
    V.tx(V.fbig, ax + 16 + 21 + 10, hy + 21, V.WHITE, at)
end

-- команды: [1] зрители, [2] T, [3] CT
function V.collect()
    local res = safe(function() return entity.get_player_resource() end, nil)
    local lp = entity.get_local_player()
    local t = {[1] = {}, [2] = {}, [3] = {}}
    for i = 1, 64 do
        local pl = safe(function() return entity.get(i) end, nil)
        if pl then
            local name = safe(function() return pl:get_name() end, nil)
            local team = safe(function() return pl.m_iTeamNum end, 0)
            if name and name ~= "" and t[team] then
                local function rv(n) return safe(function() return res[n][i] end, 0) or 0 end
                t[team][#t[team] + 1] = {
                    idx = i, ent = pl, name = name,
                    money = safe(function() return pl.m_iAccount end, 0),
                    ping = rv("m_iPing"), k = rv("m_iKills"), a = rv("m_iAssists"), d = rv("m_iDeaths"),
                    mvp = rv("m_iMVPs"), score = rv("m_iScore"),
                    alive = (team == 1) or safe(function() return pl:is_alive() end, true),
                    me = (pl == lp), spec = (team == 1),
                }
            end
        end
    end
    for _, list in pairs(t) do
        table.sort(list, function(a, b)
            if a.score ~= b.score then return a.score > b.score end
            return a.k > b.k
        end)
    end
    return t
end

function V.team_scores()
    local ct, tt = 0, 0
    for _, e in ipairs(safe(function() return entity.get_entities("CCSTeam") end, {})) do
        local n = safe(function() return e.m_iTeamNum end, 0)
        local s = safe(function() return e.m_scoreTotal end, 0)
        if n == 3 then ct = s elseif n == 2 then tt = s end
    end
    return ct, tt
end

function V.round_time()
    return safe(function()
        local gr = entity.get_game_rules()
        local left = math.max(0, gr.m_fRoundStartTime + gr.m_iRoundTime - globals.curtime)
        return string.format("%d:%02d", math.floor(left / 60), math.floor(left % 60))
    end, "--:--")
end

function V.server_name()
    local cands = {
        function() return globals.server_name end,
        function() return utils.net_channel().remote_name end,
        function() return cvar.hostname:string() end,
    }
    for _, f in ipairs(cands) do
        local v = safe(f, nil)
        if type(v) == "string" and v ~= "" then return v end
    end
    return "Server"
end

function V.draw_board(sh)
    local data = V.collect()
    local W, y0 = 720, 62
    local x = math.floor(sh.x / 2 - W / 2)
    local COL = {money = 367, ping = 415, k = 475, a = 528, d = 582, mvp = 624, score = 681}

    local secs = {
        {"COUNTER-TERRORISTS", data[3], V.CT},
        {"TERRORISTS", data[2], V.TT},
    }
    if #data[1] > 0 then secs[#secs + 1] = {"SPECTATORS", data[1], V.WHITE} end

    -- высота: заголовок секции → 29 до первой строки → шаг 33 → 34 до следующего заголовка
    local hgt = 61
    for i, s in ipairs(secs) do
        local n = #s[2]
        if i < #secs then hgt = hgt + (n > 0 and (29 + 33 * (n - 1) + 34) or 34)
        else hgt = hgt + (n > 0 and (29 + 33 * (n - 1) + 29) or 29) end
    end

    -- верхние плашки
    local map = tostring(safe(function() return globals.mapname end, "") or "")
    map = (map:gsub("^de_", ""):gsub("^cs_", ""):gsub("^%l", string.upper))
    if map == "" then map = "Match" end
    local mw = text_width(map, V.fb) + text_width("Competitive", V.fs) + 44
    V.glass(x + 1, 12, mw, 34, 1)
    V.tx(V.fb, x + 17, 29, V.WHITE, map)
    V.tx(V.fs, x + 17 + text_width(map, V.fb) + 12, 30, V.DIM, "Competitive")

    local ct_s, t_s = V.team_scores()
    local cx = sh.x / 2
    V.glass(cx - 61, 12, 122, 36, 1)
    V.tx(V.fmid, cx - 42, 30, V.CT, ct_s, 1, "c")
    V.tx(V.f, cx, 30, V.WHITE, V.round_time(), 1, "c")
    V.tx(V.fmid, cx + 42, 30, V.TT, t_s, 1, "c")

    V.glass(x + W - 112, 12, 112, 38, 1)
    V.top_icons(x + W - 112 + 16, 31)

    -- основная панель
    V.glass(x, y0, W, hgt, 1, 18, 175)

    local gr = safe(function() return entity.get_game_rules() end, nil)
    local played = safe(function() return gr.m_totalRoundsPlayed end, 0) + 1
    local maxr = safe(function() return cvar.mp_maxrounds:int() end, 30)
    if not maxr or maxr <= 0 then maxr = 30 end
    V.tx(V.fs, x + 13, y0 + 25, V.WHITE, "Server: " .. V.cut(V.server_name(), 70))
    V.tx(V.fs, x + W - 14, y0 + 25, V.WHITE, "Round " .. played .. " / " .. maxr, 1, "r")

    local cy = y0 + 61
    for _, s in ipairs(secs) do
        local title, list, accent = s[1], s[2], s[3]
        V.tx(V.fs, x + 13, cy, accent, title)
        for key, label in pairs({money = "Money", ping = "Ping", k = "K", a = "A", d = "D", mvp = "MVP", score = "Score"}) do
            V.tx(V.fs, x + COL[key], cy, V.DIM, label, 1, "c")
        end
        for k, p in ipairs(list) do
            local ry = cy + 29 + (k - 1) * 33
            local a = p.alive and 1 or 0.55
            local ac = p.alive and accent or V.DIM

            if p.me then
                render.rect(vector(x + 6, ry - 16), vector(x + W - 6, ry + 15), color(255, 255, 255, 22), 8)
            end
            render.rect(vector(x + 13, ry - 13), vector(x + 16, ry + 13),
                color(ac.r, ac.g, ac.b, math.floor(255 * a)), 1)

            -- аватарка (если API отдаёт текстуру), иначе заглушка
            local ax, ay = x + 26, ry - 14
            local drawn = false
            local tex = V.avatar(p.ent, p.idx)
            if tex then
                drawn = pcall(render.texture, tex, vector(ax, ay), vector(28, 28),
                    color(255, 255, 255, math.floor(255 * a)), "f", 4)
            end
            if not drawn then
                render.rect(vector(ax, ay), vector(ax + 28, ay + 28), color(40, 42, 52, math.floor(255 * a)), 4)
                V.tx(V.fb, ax + 14, ay + 14, V.DIM, "?", a, "c")
            end

            local nm = V.cut(p.name, 28)
            V.tx(V.f, x + 63, ry, ac, nm, a)
            if not p.alive and not p.spec then
                V.pix(x + 63 + text_width(nm, V.f) + 8, ry - 7, 2, V.ICON.skull, V.DIM)
            end

            V.tx(V.fs, x + COL.money, ry, V.WHITE, "$" .. p.money, a, "c")
            V.tx(V.fs, x + COL.ping, ry, V.WHITE, p.spec and "SPEC" or tostring(p.ping), a, "c")
            V.tx(V.fs, x + COL.k, ry, V.WHITE, p.k, a, "c")
            V.tx(V.fs, x + COL.a, ry, V.WHITE, p.a, a, "c")
            V.tx(V.fs, x + COL.d, ry, V.WHITE, p.d, a, "c")
            if p.mvp > 0 then V.tx(V.fs, x + COL.mvp, ry, V.WHITE, p.mvp, a, "c") end
            V.tx(V.fs, x + COL.score, ry, V.WHITE, p.score, a, "c")
        end
        cy = cy + (#list > 0 and (29 + 33 * (#list - 1) + 34) or 34)
    end

    -- нижняя плашка: раунды и зрители
    local by = y0 + hgt + 16
    V.glass(x, by, W, 34, 1)
    V.tx(V.fs, x + 30, by + 17, V.DIM, "Rounds")
    local last = played - 1
    local first = last > 15 and 16 or 1
    for r = first, math.min(last, first + 14) do
        V.tx(V.fs, x + 83 + (r - first) * 25, by + 17, V.DIM, r, 1, "c")
    end
    V.tx(V.fs, x + W - 20, by + 17, V.DIM, "Spectators: " .. #data[1], 1, "r")
end

-- прячем строки стандартного чата, пока включён кастомный
function V.apply_cvar()
    local want = M.vchat:get() and M.enable:get()
    if want ~= V.applied then
        V.applied = want
        pcall(function() cvar.hud_saytext_time:float(want and 0 or 12) end)
    end
end

function V.draw()
    local sh = render.screen_size()
    local lp = entity.get_local_player()
    if M.vchat:get() then
        if M.vchat_input:get() then V.poll_chat() end
        V.draw_chat(sh)
    end
    if M.vhud:get() and lp and lp:is_alive() then V.draw_hud(lp, sh) end
    if M.vboard:get() and V.board_down() then V.draw_board(sh) end
end

    if not M.enable:get() then return end

    V.draw()

end)

local function reset_all()
    for _, ref in pairs(R) do ovr(ref) end
    for _, ref in ipairs(hc_refs) do ovr(ref) end
    for _, ref in ipairs(H.hb) do ovr(ref) end
    H.hb_on = false

end

        return

    if not M.anim:get() then ovr(R.legs) return end

        if ov and lp.m_vecVelocity:length2d() >= 3 then ov.weight = 1 end

        local v, g = M.anim_value:get(), M.anim_ground:get()

        elseif g == "Jitter" then lp.m_flPoseParameter[0] = (globals.tickcount % 4 > 1) and v / 10 or 0
        else lp.m_flPoseParameter[0] = math.random(v, 10) / 10 end

        local a = M.anim_air:get()
        if a == "Static" then lp.m_flPoseParameter[6] = 1
        elseif a == "Kangaroo" then lp.m_flPoseParameter[6] = math.random(0, 10) / 10 end
    end)
end)

local function pick_phrase(mode)
    local list
    if mode == "RU" then list = PHRASES_RU
    elseif mode == "EN" then list = PHRASES_EN
    else
        list = {}
        for _, p in ipairs(PHRASES_RU) do list[#list + 1] = p end
        for _, p in ipairs(PHRASES_EN) do list[#list + 1] = p end
    end
    return list[math.random(1, #list)]
end

on("player_death", function(e)
    local lp = entity.get_local_player()
    local victim, attacker = entity.get(e.userid, true), entity.get(e.attacker, true)

    if not M.enable:get() then return end
    if attacker ~= lp or not victim or not victim:is_enemy() then return end

    local mode, text = M.killsay:get(), nil
    if mode == "1" then text = "1"
    elseif mode ~= "None" then text = pick_phrase(mode) end
    if text then
        local delay = mode == "1" and 0.5 or (1.2 + math.random() * 1.4)
        pcall(utils.execute_after, delay, function()
            utils.console_exec('say "' .. text .. '"')
        end)
    end
end)
